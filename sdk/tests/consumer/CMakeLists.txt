cmake_minimum_required(VERSION 3.25)
project(player_sdk_external_consumer LANGUAGES CXX)
if(NOT PLAYER_SDK_SOURCE_DIR)
    message(FATAL_ERROR "Set PLAYER_SDK_SOURCE_DIR to an absolute SDK source path")
endif()
set(BUILD_SHARED_LIBS ON)
set(BUILD_TESTING ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/host-bin")
set(CMAKE_CXX_STANDARD 17)
set(saved_runtime "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
set(saved_crt "${CMAKE_MSVC_RUNTIME_LIBRARY}")
if(PLAYER_SDK_BUILD_QT)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
    qt_standard_project_setup(REQUIRES 6.8)
endif()
add_subdirectory("${PLAYER_SDK_SOURCE_DIR}" sdk)
if(NOT BUILD_SHARED_LIBS OR NOT BUILD_TESTING OR NOT CMAKE_CXX_STANDARD EQUAL 17
    OR NOT "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}" STREQUAL "${saved_runtime}"
    OR NOT "${CMAKE_MSVC_RUNTIME_LIBRARY}" STREQUAL "${saved_crt}")
    message(FATAL_ERROR "SDK modified host configuration")
endif()
if(TARGET myplayer OR TARGET myplayer_monitor OR TARGET play_core_refactor_tests)
    message(FATAL_ERROR "SDK injected applications or private tests")
endif()
get_target_property(core_includes play_core INTERFACE_INCLUDE_DIRECTORIES)
foreach(path IN LISTS core_includes)
    if(NOT path MATCHES "/core/include$")
        message(FATAL_ERROR "Private include leaked: ${path}")
    endif()
endforeach()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE PlayerSDK::Core)
if(PLAYER_SDK_BUILD_QT)
    qt_add_executable(qt_consumer qt_main.cpp)
    target_link_libraries(qt_consumer PRIVATE PlayerSDK::QtWidgets)
endif()
if(WIN32)
    add_custom_command(TARGET consumer POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:consumer> $<TARGET_FILE_DIR:consumer>
        COMMAND_EXPAND_LISTS
    )
endif()
